@Override
public User login(LoginRequest request) {

    String username = request.getUsername();

    // 1. Check if the username is blocked
    if (loginRateLimitor.isBlocked(username)) {
        throw new IllegalArgumentException(
                "Too many login attempts. Try again after 15 minutes."
        );
    }

    // 2. Find user and record failure for unknown usernames
    User user = userRepository.findByUsername(username)
            .orElseThrow(() -> {
                loginRateLimitor.loginFailed(username);
                return new IllegalArgumentException(
                        "Invalid username or password"
                );
            });

    // 3. Check email verification
    if (!user.isEmailVerified()) {
        throw new IllegalArgumentException(
                "Please verify your email first"
        );
    }

    // 4. Check account status
    if (!user.isActive()) {
        throw new IllegalArgumentException(
                "User account is inactive"
        );
    }

    // 5. Verify password
    if (!passwordEncoder.matches(
            request.getPassword(), user.getPassword())) {

        loginRateLimitor.loginFailed(username);

        throw new IllegalArgumentException(
                "Invalid username or password"
        );
    }

    // 6. Clear failed attempts after successful login
    loginRateLimitor.loginSucceeded(username);

    return user;
}
