package com.expensetracker.api.service;

import com.expensetracker.api.DTO.MonthlySummaryResponse;
import com.expensetracker.api.model.BudgetAlert;
import com.expensetracker.api.model.BudgetAlertType;
import com.expensetracker.api.model.CategoryType;
import com.expensetracker.api.model.User;
import com.expensetracker.api.repository.BudgetAlertRepository;
import com.expensetracker.api.service.impl.BudgetAlertServiceImpl;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;

import java.math.BigDecimal;
import java.time.Month;
import java.time.Year;

import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class BudgetAlertServiceTest {

    @Mock
    private MonthlySummaryService monthlySummaryService;

    @Mock
    private CurrentUserService currentUserService;

    @Mock
    private EmailService emailService;

    @Mock
    private BudgetAlertRepository budgetAlertRepository;

    @InjectMocks
    private BudgetAlertServiceImpl budgetAlertService;

    private final Month month = Month.OCTOBER;
    private final Year year = Year.of(2026);
    private final CategoryType category = CategoryType.Food;

    private User createUser() {
        User user = new User();
        user.setId(1);
        user.setUsername("tej");
        user.setMail("tej@example.com");
        return user;
    }

    private MonthlySummaryResponse createSummary(
            String budget,
            String spent
    ) {
        return new MonthlySummaryResponse(
                category,
                new BigDecimal(budget),
                new BigDecimal(spent)
        );
    }

    @Test
    void shouldSend80PercentWarningOnlyOnce() {
        User user = createUser();

        when(currentUserService.getCurrentUser())
                .thenReturn(user);

        when(monthlySummaryService.getSummaryOfCategory(
                month, year, category
        )).thenReturn(createSummary("5000", "4000"));

        // First check: not sent.
        // Second check: already sent.
        when(budgetAlertRepository
                .existsByUserAndCategoryAndMonthAndYearAndAlertType(
                        user, category, month, 2026,
                        BudgetAlertType.WARNING_80_PERCENT
                )).thenReturn(false, true);

        budgetAlertService.checkBudgetAlert(month, year, category);
        budgetAlertService.checkBudgetAlert(month, year, category);

        verify(emailService, times(1))
                .sendEmail(
                        eq("tej@example.com"),
                        contains("Budget warning"),
                        anyString()
                );

        verify(budgetAlertRepository, times(1))
                .save(any(BudgetAlert.class));
    }

    @Test
    void shouldSend100PercentAlertOnlyOnce() {
        User user = createUser();

        when(currentUserService.getCurrentUser())
                .thenReturn(user);

        when(monthlySummaryService.getSummaryOfCategory(
                month, year, category
        )).thenReturn(createSummary("5000", "5200"));

        when(budgetAlertRepository
                .existsByUserAndCategoryAndMonthAndYearAndAlertType(
                        user, category, month, 2026,
                        BudgetAlertType.EXCEEDED_100_PERCENT
                )).thenReturn(false, true);

        budgetAlertService.checkBudgetAlert(month, year, category);
        budgetAlertService.checkBudgetAlert(month, year, category);

        verify(emailService, times(1))
                .sendEmail(
                        eq("tej@example.com"),
                        contains("Budget limit reached"),
                        anyString()
                );

        verify(budgetAlertRepository, times(1))
                .save(any(BudgetAlert.class));
    }

    @Test
    void shouldSendBothAlertsAtTheirRespectiveThresholds() {
        User user = createUser();

        when(currentUserService.getCurrentUser())
                .thenReturn(user);

        when(monthlySummaryService.getSummaryOfCategory(
                month, year, category
        )).thenReturn(
                createSummary("5000", "4000"),
                createSummary("5000", "5000")
        );

        when(budgetAlertRepository
                .existsByUserAndCategoryAndMonthAndYearAndAlertType(
                        user, category, month, 2026,
                        BudgetAlertType.WARNING_80_PERCENT
                )).thenReturn(false);

        when(budgetAlertRepository
                .existsByUserAndCategoryAndMonthAndYearAndAlertType(
                        user, category, month, 2026,
                        BudgetAlertType.EXCEEDED_100_PERCENT
                )).thenReturn(false);

        budgetAlertService.checkBudgetAlert(month, year, category);
        budgetAlertService.checkBudgetAlert(month, year, category);

        verify(emailService, times(1))
                .sendEmail(
                        eq("tej@example.com"),
                        contains("Budget warning"),
                        anyString()
                );

        verify(emailService, times(1))
                .sendEmail(
                        eq("tej@example.com"),
                        contains("Budget limit reached"),
                        anyString()
                );

        verify(budgetAlertRepository, times(2))
                .save(any(BudgetAlert.class));
    }
}
